**命令 7 —— 复现 §2.3 的锚计数（三层口径）**

```bash
cd "/workspace/Astro CS Database"
python3 - <<'PY'
import os,re,collections
ROOT=os.getcwd(); DOCS=os.path.join(ROOT,"docs")
idx=collections.defaultdict(list)
for base in ("lib","eng"):
    for dp,dn,fn in os.walk(os.path.join(ROOT,base)):
        if "__pycache__" in dp: continue
        for f in fn: idx[f].append(os.path.relpath(os.path.join(dp,f),ROOT))
pat=re.compile(r'([A-Za-z0-9_./-]+\.(?:cpp|hpp|h|cc|py|cmake|json|yaml|yml|ini|txt)):(\d+)(?:-(\d+))?')
lc={}
def L(p):
    if p not in lc:
        try: lc[p]=sum(1 for _ in open(os.path.join(ROOT,p),'rb'))
        except Exception: lc[p]=None
    return lc[p]
res=collections.defaultdict(set); occ=collections.Counter(); tot=0
for dp,dn,fn in os.walk(DOCS):
    for f in fn:
        if not f.endswith(('.md','.yaml','.yml')): continue
        p=os.path.join(dp,f); rel=os.path.relpath(p,ROOT)
        for ln,line in enumerate(open(p,encoding='utf-8',errors='replace'),1):
            for m in pat.finditer(line):
                path,num=m.group(1),int(m.group(2)); tot+=1
                cands=[path] if os.path.isfile(os.path.join(ROOT,path)) else idx.get(os.path.basename(path),[])
                k=(rel,ln,path,num)
                if not cands: res['a_notfound'].add(k); occ['a_notfound']+=1
                elif len(cands)>1: res['e_ambiguous'].add(k); occ['e_ambiguous']+=1
                else:
                    n=L(cands[0])
                    if n is None: res['a_notfound'].add(k); occ['a_notfound']+=1
                    elif num>n: res['b_outofrange'].add(k); occ['b_outofrange']+=1
                    else: res['d_inrange'].add(k); occ['d_inrange']+=1
for k in sorted(occ): print(f"  口径I {k:14s} {occ[k]:5d}")
print("  口径I TOTAL", tot)
for k in sorted(res): print(f"  口径II {k:14s} {len(res[k]):5d}")
t2=sum(len(v) for v in res.values())
m=len(res['a_notfound'])+len(res['e_ambiguous'])+len(res['b_outofrange'])
print(f"  口径II TOTAL {t2}  | 口径III 整改分母 {m} ({100.0*m/t2:.1f}%)  | 口径IV 须人读 {len(res['d_inrange'])} ({100.0*len(res['d_inrange'])/t2:.1f}%)")
bare=0; pat2=re.compile(r'(^|[^A-Za-z0-9_./-]):[0-9]+')
for dp,dn,fn in os.walk(DOCS):
    for f in fn:
        if f.endswith(('.md','.yaml','.yml')):
            bare+=len(pat2.findall(open(os.path.join(dp,f),encoding='utf-8',errors='replace').read()))
print(f"  裸相对锚 {bare} | 锚总量下界 {tot+bare}")
PY
```
