## 5. 交付验收器（本单最有价值的产物）

### 5.1 三处非显然的实现要点

**① 为什么 C3 不能只用 `yaml.safe_load`。** 重复映射键在 PyYAML 下是**静默吞并**：`dict[k] = v` 逐对赋值，重复键**末值胜**。本单的缺陷在 `safe_load` 下**不可见**——它安静返回被吞并后的值，不报任何错。所以 C3 自建 Loader：先扫节点内全部 `(key_node, value_node)` 对、命中重复即记录**首值行号与重复值行号**，再按 PyYAML 语义（末值胜）构造数据，保证返回结构与 `safe_load` 一致、只多出重复键报告。

**② 为什么 C2 的 duty 检查必须按文本取值。** 同上机制：`production_call_paths_stage2.csv` 的退化 duty 被紧随其后的孤儿字段块吞并，任何基于解析结果的扫描都看不到它（见 §1.4）。故 duty 子判据**按文本逐行**取 `- path:` 与其后的 `duty:`，不用解析结果。

**③ 为什么 C4 的标题解析要按括号配平。** `§5.4` 的真实标题「天光平面（UPM）」自身含全角括号，非配平正则会把它截成「天光平面」造成假红。

另外，`DOCUMENT_GOVERNANCE.md:56` 规定裸名 `ACSD_DESIGN.md` 在 `docs/` 内是**正确写法**、在 `docs/` 外才带 `docs/` 前缀。C4 因此**两种形态都认**——只认带前缀的形式会漏掉整片抬头（本单旗舰缺陷正好全在裸名抬头里，差点漏网）。

def load_yaml_strict(text: str):
    """返回 (data, dup_list)。dup_list 元素 = (mapping 路径, 键, 首值行号, 末值行号)。

    实现要点：SafeLoader 默认的 construct_mapping 是 `dict[k] = v`，重复键末值静默覆盖首值。
    这里改成先扫节点内所有 (key_node, value_node) 对，命中重复即记录并报出两侧行号，
    再按 PyYAML 语义（末值胜）构造数据，保证返回结构与 safe_load 一致、只多出重复键报告。
    """
    import yaml

    class StrictLoader(yaml.SafeLoader):
        pass

    dups: list[tuple[str, str, int, int]] = []

    def _construct_mapping(loader, node, deep=False):
        seen: dict[str, tuple[int, object]] = {}
        for key_node, value_node in node.value:
            key = loader.construct_object(key_node, deep=deep)
            if key in seen:
                first_line = seen[key][0]
                dups.append((_node_path(loader, node), str(key), first_line,
                             key_node.start_mark.line + 1))
                # 末值胜：与 PyYAML 原行为一致
                loader.construct_mapping(node, deep=deep)
            else:
                seen[key] = (key_node.start_mark.line + 1, value_node)
        return yaml.SafeLoader.construct_mapping(loader, node, deep=deep)

    def _node_path(loader, node) -> str:
        # 尽力给出映射的可读位置；拿不到就退回行号
        return f"<mapping@line {node.start_mark.line + 1}>"

    StrictLoader.add_constructor(
        yaml.resolver.BaseResolver.DEFAULT_MAPPING_TAG, _construct_mapping)
    data = yaml.load(text, Loader=StrictLoader)
    return data, dups

def design_sections(design_text: str) -> dict[str, str]:
    """节号 -> 标题。覆盖 ## / ### 两级；附录按「附录 A」「附录 B」分别建键，
    否则两个附录会塌成同一个键而少计一节。"""
    out: dict[str, str] = {}
    for line in design_text.splitlines():
        m = re.match(r"^(#{2,3})\s+(.*)$", line)
        if not m:
            continue
        rest = m.group(2).strip()
        ap = re.match(r"^(附录\s*[A-Z])\.?\s+(.*)$", rest)
        if ap:
            out[ap.group(1).replace(" ", "")] = ap.group(2).strip()
            continue
        num, _, title = rest.partition(" ")
        num = num.rstrip(".、")
        title = title.strip()
        if num:
            out[num] = title
    return out


def norm_title(s: str) -> str:
    """去空白后逐字比较（§4.5）。"""
    return re.sub(r"\s+", "", s or "")

DESIGN_SEG_RE = re.compile(r"docs/ACSD_DESIGN\.md(?P<seg>(?:[^；;]*))")
# 两种合法引用形态：§8.4（顶层结构）与 附录 A（术语）。
# 数值节号必须带 § —— 否则正文里的 "MSVC v143" 之类裸数字会被误读成节号。
_NUM = r"\d+(?:\.\d+)*(?:\s*/\s*§?\s*\d+(?:\.\d+)*)?"
_SEC_HEAD_RE = re.compile(
    r"(?:§\s*(?P<num>" + _NUM + r")|附录\s*(?P<apx>[A-Z]))"
)


def _balanced_title(text: str, start: int):
    """从 start（指向「（」）起按全角括号配平取出标题，返回 (标题, 结束位置)。

    最高设计 §5.4 的真实标题是「天光平面（UPM）」，本身含全角括号；
    非配平的正则会把它截成「天光平面」，导致逐字比对假红。
    """
    if start >= len(text) or text[start] != "（":
        return None, start
    depth = 0
    for i in range(start, len(text)):
        if text[i] == "（":
            depth += 1
        elif text[i] == "）":
            depth -= 1
            if depth == 0:
                return text[start + 1:i], i + 1
    return None, start


def parse_design_refs(upstream: str) -> list[tuple[str, str | None]]:
    """从 upstream 串中抽出指向最高设计的 (节号, 括注标题|None)。

    两种路径形态都算：带 `docs/` 前缀与裸名 `ACSD_DESIGN.md`。裸名在 docs/ 内是正确写法
    （DOCUMENT_GOVERNANCE.md §4.1.2「裸名 ACSD_DESIGN.md 在 docs/ 外带 docs/ 前缀、
    在 docs/ 内保持裸名」），只认带前缀的形式会漏掉整片抬头。

    只取从 ACSD_DESIGN.md 起到下一个文档路径引用为止的片段，避免把
    docs/science/... §1 这类指向别的文档的节号误判为最高设计节号。
    """
    refs: list[tuple[str, str | None]] = []
    parts = re.split(r"(?=(?:docs|eng|lib|artifacts|实验|testdata)/[\w./-]*[\w/])",
                     upstream)
    for part in parts:
        mm = re.search(r"(?:docs/)?ACSD_DESIGN\.md", part)
        if not mm:
            continue
        seg = part[mm.end():]
        pos = 0
        while True:
            m = _SEC_HEAD_RE.search(seg, pos)
            if not m:
                break
            # 跳过 § 与节号之间的空白
            tpos = m.end()
            while tpos < len(seg) and seg[tpos] in " \t":
                tpos += 1
            title, nxt = _balanced_title(seg, tpos)
            if title is None:
                nxt = m.end()
            if m.group("apx"):
                base_nums = ["附录" + m.group("apx")]
            else:
                base_nums = [n.strip() for n in re.split(r"\s*/\s*§?\s*", m.group("num"))]
            for n in base_nums:
                refs.append((n.replace(" ", ""), title))
            pos = nxt
    return refs

    # 必须按**文本**逐行取 duty，不能用解析结果：重复映射键下 PyYAML 末值胜会把真实的

    # 退化 duty 被紧随其后的孤儿字段块吞并，safe_load 完全看不到）。
    cur_path = None
    for ln_no, raw in enumerate(index_text.splitlines(), 1):
        mp = re.match(r'^\s*- path:\s*"([^"]+)"', raw)
        if mp:
            cur_path = mp.group(1)
            continue
        md = re.match(r'^\s*duty:\s*"([^"]*)"\s*$', raw)
        if md and cur_path and md.group(1).strip() == cur_path:
            v.append({"where": f"DOCUMENT_INDEX.yaml:{ln_no}",
                      "issue": f"duty 退化：duty 直接写成自身路径「{cur_path}」，未描述职责",
                      "path": cur_path})
    reports["C2"] = {
        "name": "tree-coverage",
        "ok": not v,
        "detail": f"文档层对象 {len(doc_layer)} ＝ 判定对象 {len(judged)} ＋ 排除面命中 {len(excluded)}；"
                  f"判红 {len(v)} 条",
        "exclusion_surface": [
            {"pattern": x["pattern"], "reason": x["reason"], "evidence": x["evidence"],
             "hits": len(excluded) if x is EXCLUSIONS[0] else 0}
            for x in EXCLUSIONS
        ],
        "violations": v,
    }

    # ---------------- C4：upstream 双要素（节号存在 + 标题逐字相符） ----------------
    sections = design_sections(design_text)
    if not sections:
        res["checks"]["C4"] = {"ok": False,
                               "error": "fail-closed：未从最高设计解析出任何节号，"
                                        "判据输入面为空（DOCUMENT_GOVERNANCE.md §4.3 非空断言）"}
        sections = {}
    v = []
    for zone, i, e in entries:
        up = (e or {}).get("upstream")
        if not up:
            continue
        for num, title in parse_design_refs(str(up)):
            base = f"DOCUMENT_INDEX.yaml doc_index.{zone}[{i}] ({ent_path(e)})"
            if num not in sections:
                v.append({"where": base,
                          "issue": f"节号悬空：docs/ACSD_DESIGN.md 不存在 §{num}",
                          "sec": num, "title": title})
                continue
            if title is None:
                v.append({"where": base,
                          "issue": f"只给号不给标题：§{num} 缺「（该节标题）」双要素",
                          "sec": num, "title": None})
                continue
            actual = sections[num]
            if norm_title(title) != norm_title(actual):
                v.append({"where": base,
                          "issue": f"标题错位：§{num} 括注「{title}」≠ 实际标题「{actual}」",
                          "sec": num, "title": title, "actual_title": actual})
    # 文档抬头的双要素同样校验（§4.5：文档抬头与索引条目同源；标题错位类缺陷住在抬头里）
    hdr_hits = 0
    doc_up = sorted(p for p in doc_layer
                    if p.startswith("docs/")
                    and re.search(r"ACSD_DESIGN\.md\s*§", open(
                        os.path.join(root, p), encoding="utf-8").read()[:4000]))
    for p in doc_up:
        txt = open(os.path.join(root, p), encoding="utf-8").read()[:4000]
        for ln_no, line in enumerate(txt.splitlines(), 1):
            if "ACSD_DESIGN.md" not in line or "§" not in line:
                continue
            for num, title in parse_design_refs(line):
                hdr_hits += 1
                base = f"{p}:{ln_no}"
                if num not in sections:
                    v.append({"where": base, "issue": f"抬头节号悬空：不存在 §{num}",
                              "sec": num, "title": title})
                elif title is not None and norm_title(title) != norm_title(sections[num]):
                    v.append({"where": base,
                              "issue": f"抬头标题错位：§{num} 括注「{title}」≠ 实际标题「{sections[num]}」",
                              "sec": num, "title": title, "actual_title": sections[num]})
    reports["C4"] = {
        "name": "upstream-two-element",
        "ok": not v,
        "detail": f"设计节号表 {len(sections)} 节；索引条目与文档抬头合计命中 {hdr_hits} 处抬头引用；"
                  f"判红 {len(v)} 条",
        "violations": v,
    }

    # ---------------- C5：跨文档合同 ID 存在于 docs/contracts/INDEX.yaml ----------------
    try:
        cdoc, _ = load_yaml_strict(open(os.path.join(root, contracts_rel),
                                        encoding="utf-8").read())
    except Exception as e:
        res["checks"]["C5"] = {"ok": False, "error": f"合同索引解析失败: {e}"}
        cdoc = {}
    known = set()
    for c in ((cdoc or {}).get("contracts") or []):
        if isinstance(c, dict) and c.get("id"):
            known.add(str(c["id"]))
    legacy = ((cdoc or {}).get("legacy_contract_id_map") or [])
    if isinstance(legacy, dict):
        known |= {str(k) for k in legacy}
    elif isinstance(legacy, list):
        for item in legacy:
            if isinstance(item, dict) and item.get("legacy_id"):
                known.add(str(item["legacy_id"]))
    if not known:
        res["checks"]["C5"] = {"ok": False,
                               "error": "fail-closed：合同注册表解析出 0 个 ID，判据输入面为空"}
        known = set()

    # 引用面：正式文档层（docs/**/*.md，排除索引自身与目录招牌件）
    scanned = [p for p in doc_layer if p != index_rel and not excl_re.match(p)]
    refs: dict[str, set[str]] = {}
    for p in scanned:
        for ln_no, line in enumerate(open(os.path.join(root, p), encoding="utf-8"),
                                      1):
            for tok in CONTRACT_ID_RE.findall(line):
                if tok in known or tok.endswith(PLACEHOLDER_SUFFIX):
                    continue
                refs.setdefault(tok, set()).add(f"{p}:{ln_no}")
    v = [{"issue": f"合同 ID「{k}」被引用 {len(s)} 处，但不在 docs/contracts/INDEX.yaml "
                  f"（contracts[].id ∪ legacy_contract_id_map[].legacy_id）",
          "id": k, "where": sorted(s)[:3], "occurrences": len(s)}
         for k, s in sorted(refs.items())]
    by_layer: dict[str, int] = {}
    for x in v:
        lay = str(x["id"]).split("-", 1)[0]
        by_layer[lay] = by_layer.get(lay, 0) + 1
    reports["C5"] = {
        "name": "contract-id-alive",
        "ok": not v,
        "detail": f"在册合同 ID {len(known)} 个（注册面 docs/contracts/INDEX.yaml）；"
                  f"引用面扫描 {len(scanned)} 份文档；"
                  f"悬空合同 ID {len(v)} 个，按层 {dict(sorted(by_layer.items()))}",
        "id_grammar_source": "docs/engineering/TRACEABILITY_SPEC.md §3（八层注册正则）",
        "violations": v,
    }

    for k in ("C1", "C2", "C3", "C4", "C5"):
        reports[k]["check_id"] = k
    res["checks"] = reports
    return res

SELF_TEST_DEFECTS = {
    "C1": "把某条 path 改成不存在的悬空路径",
    "C2": "删掉一条文档的索引登记 / 把 duty 改成自身路径",
    "C3": "在一条正常条目后追加无 path 键的孤儿字段块",
    "C4": "把 upstream 的节号改成设计里不存在的节号",
    "C5": "把 INDEX.yaml 里的某条合同 id 改名",
}


def self_test(root: str) -> int:
    """注入缺陷 → 判据必须转红。

    在临时的最小仓（只含 docs/，git init + add，足够支撑 `git ls-files` 派生对象集）
    里做注入，不碰真实工作树：并行车道随时在读这些文件，就地改会污染它们的读数。
    """
    import shutil
    base = run_checks(root)
    if not all(c.get("ok") for c in base["checks"].values()):
        print("self-test 基线不绿（本仓 C4/C5 仍有未清缺陷，属预期）。"
              "注入后判红只应更多：")
        for k, c in base["checks"].items():
            print(f"  {k} {c.get('name')}: ok={c.get('ok')} "
                  f"violations={len(c.get('violations', []))}")
    tmp = tempfile.mkdtemp(prefix="g08_03a_selftest_")
    try:
        work = os.path.join(tmp, "repo")
        os.makedirs(work)
        shutil.copytree(os.path.join(root, "docs"), os.path.join(work, "docs"),
                        symlinks=True)
        subprocess.run(["git", "init", "-q"], cwd=work, check=True)
        subprocess.run(["git", "add", "-A"], cwd=work, check=True)
        idx = os.path.join(work, "docs/DOCUMENT_INDEX.yaml")
        cidx = os.path.join(work, "docs/contracts/INDEX.yaml")
        text = open(idx, encoding="utf-8").read()
        results = {}

        def probe(key):
            results[key] = run_checks(work)["checks"][key]

        # C1 注入：把 path 改成不存在的悬空路径
        t1 = text.replace('- path: "docs/GLOSSARY.md"',
                          '- path: "docs/__NO_SUCH_FILE__.md"', 1)
        assert t1 != text, "C1 注入锚点未命中"
        open(idx, "w", encoding="utf-8").write(t1)
        probe("C1")
        open(idx, "w", encoding="utf-8").write(text)

        # C2 注入：删掉一条 docs 文档的整条索引登记
        keep, skip = [], False
        for ln in text.splitlines(keepends=True):
            if '- path: "docs/detail/common.md"' in ln:
                skip = True
                continue
            if skip:
                if ln.startswith("    - path:") or ln.startswith("  archived:"):
                    skip = False
                else:
                    continue
            keep.append(ln)
        open(idx, "w", encoding="utf-8").write("".join(keep))
        c2a = run_checks(work)["checks"]["C2"]

        # C2 注入（2/3）：把一条 duty 改成自身路径（退化 duty）
        m = re.search(r'^(\s*)- path:\s*"([^"]+)"$(.*?)^\s*duty:\s*"([^"]+)"\s*$',
                      text, re.M | re.S)
        assert m, "C2 duty 注入锚点未命中：索引里没有 path+duty 配对"
        t2b = text[:m.start()] + (f'{m.group(1)}- path: "{m.group(2)}"'
                                  + m.group(3) + f'{m.group(1)}  duty: "{m.group(2)}"') \
            + text[m.end():]
        assert f'duty: "{m.group(2)}"' in t2b, "C2 duty 注入未生效"
        open(idx, "w", encoding="utf-8").write(t2b)
        c2b = run_checks(work)["checks"]["C2"]
        results["C2"] = {"name": c2b["name"], "ok": c2b["ok"],
                         "violations": [x for x in c2b["violations"]
                                        if "duty 退化" in x["issue"]]}
        if not results["C2"]["violations"]:
            results["C2"] = c2b          # 没抓到就用整份报告，至少能看出判红
        open(idx, "w", encoding="utf-8").write(text)
        _ = c2a

        # C3 注入：孤儿字段块（无 - path: 键；PyYAML 末值胜会静默吞并首值）
        t3 = text.replace(
            '      downstream: "CMakeLists.txt、docchk002_claims_fixture.csv、test_linux_release.py 等 9 处"',
            '      downstream: "CMakeLists.txt、docchk002_claims_fixture.csv、test_linux_release.py 等 9 处"\n'
            '      status: ACTIVE_INFORMATIVE\n'
            '      duty: "注入的孤儿字段块"\n'
            '      upstream: "docs/ACSD_DESIGN.md §8（软件架构）"\n'
            '      downstream: ""', 1)
        assert t3 != text, "C3 注入锚点未命中"
        open(idx, "w", encoding="utf-8").write(t3)
        probe("C3")
        open(idx, "w", encoding="utf-8").write(text)

        # C4 注入：节号改成设计里不存在的节号
        t4 = text.replace('upstream: "docs/ACSD_DESIGN.md §13（版本与发布权）"',
                          'upstream: "docs/ACSD_DESIGN.md §99.9（不存在的节）"', 1)
        assert t4 != text, "C4 注入锚点未命中"
        open(idx, "w", encoding="utf-8").write(t4)
        probe("C4")
        open(idx, "w", encoding="utf-8").write(text)

        # C5 注入：改掉一个在册合同 ID
        ctext = open(cidx, encoding="utf-8").read()
        t5 = ctext.replace("  - id: SCI-SCOPE-001", "  - id: SCI-SCOPE-XXX", 1)
        assert t5 != ctext, "C5 注入锚点未命中"
        open(cidx, "w", encoding="utf-8").write(t5)
        probe("C5")
        open(cidx, "w", encoding="utf-8").write(ctext)

        print("self-test：注入缺陷 → 判据必须转红（红 = 判据有判别力，不是恒真）")
        bad = 0
        for k in ("C1", "C2", "C3", "C4", "C5"):
            c = results[k]
            fired = not c.get("ok")
            print(f"  {k} {c.get('name')}: 注入「{SELF_TEST_DEFECTS[k]}」→ "
                  f"{'红 ✓' if fired else '绿 ✗（判据恒真！）'} "
                  f"violations={len(c.get('violations', []))}")
            if not fired:
                bad += 1
        return 1 if bad else 0
    finally:
        shutil.rmtree(tmp, ignore_errors=True)

def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description="索引与双向可追溯五判据校验器")
    ap.add_argument("--root", default=".", help="仓库根（默认当前目录）")
    ap.add_argument("--json-out", help="把完整报告写到此 JSON 文件")
    ap.add_argument("--quiet", action="store_true", help="只打印总判定与计数")
    ap.add_argument("--self-test", action="store_true", help="注入缺陷验证判据判红能力")
    a = ap.parse_args(argv)

    root = os.path.abspath(a.root)
    if a.self_test:
        return self_test(root)

    try:
        res = run_checks(root)
    except Exception as e:
        print(f"内部错误（fail-closed）: {e}", file=sys.stderr)
        return 2

    if a.json_out:
        with open(a.json_out, "w", encoding="utf-8") as f:
            json.dump(res, f, ensure_ascii=False, indent=2)

    if not a.quiet:
        for k in ("C1", "C2", "C3", "C4", "C5"):
            c = res["checks"].get(k, {})
            if "error" in c:
                print(f"[{k}] {c.get('name', k)}: ERROR {c['error']}")
                continue
            mark = "PASS" if c["ok"] else "FAIL"
            print(f"[{k}] {c['name']}: {mark} — {c['detail']}")
            if not c["ok"]:
                for i, x in enumerate(c["violations"][:60], 1):
                    print(f"    {i}. {x.get('where')} :: {x.get('issue')}")
                if len(c["violations"]) > 60:
                    print(f"    … 其余 {len(c['violations']) - 60} 条见 --json-out")
            if k == "C2" and not c["ok"]:
                for x in c.get("exclusion_surface", []):
                    print(f"    排除面 {x['pattern']} reason={x['reason']} 命中 {x['hits']}")
    else:
        for k in ("C1", "C2", "C3", "C4", "C5"):
            c = res["checks"].get(k, {})
            print(f"{k} {c.get('name', k)}: "
                  f"{'PASS' if c.get('ok') else 'FAIL'} "
                  f"({len(c.get('violations', []))} violations)")

    all_ok = all(res["checks"].get(k, {}).get("ok") for k in ("C1", "C2", "C3", "C4", "C5"))
    print("TOTAL:", "PASS" if all_ok else "FAIL")
    return 0 if all_ok else 1


if __name__ == "__main__":
    sys.exit(main())
