179: // 判据 = 中心化点云 Gram 矩阵 [[sxx,sxy],[sxy,syy]] 的特征值比 λlo/λhi,

181: // 原绝对阈值 |det|>1e-24 (量纲 px⁴) 已废除: 共线点云 det==0 时它不阻止
182: // has_spatial_field=1; 近共线点云 det 虽 >1e-24, 平面仍把方差场病态外推 ±62%。
183: constexpr double kPlaneGeomRatio = 0.0625;  // λlo/λhi 下限 (=1/16, κ≤4)
184: double plane_geometry_ratio(const double* xs, const double* ys, std::size_t n) {
185:     if (!xs || !ys || n < 2) return 0.0;
186:     double mx = 0.0, my = 0.0;
187:     for (std::size_t i = 0; i < n; ++i) { mx += xs[i]; my += ys[i]; }
188:     mx /= (double)n; my /= (double)n;
189:     double sxx = 0.0, sxy = 0.0, syy = 0.0;
190:     for (std::size_t i = 0; i < n; ++i) {
191:         const double X = xs[i] - mx;
192:         const double Y = ys[i] - my;
193:         sxx += X * X; sxy += X * Y; syy += Y * Y;
194:     }
195:     const double tr = sxx + syy;
196:     const double disc = std::sqrt((sxx - syy) * (sxx - syy) + 4.0 * sxy * sxy);
197:     const double lam_hi = 0.5 * (tr + disc);
198:     const double lam_lo = 0.5 * (tr - disc);
199:     if (!(lam_hi > 0.0) || !std::isfinite(lam_hi) || !std::isfinite(lam_lo)) return 0.0;
200:     return lam_lo / lam_hi;
