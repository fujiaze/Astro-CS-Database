# REAL-SCIENCE-001 测量驱动构建（write_scope: run/v6/real-science/）
# 直接编译冻结 V6 低层权威源（PSF/W_info/PSFSW），不修改任何生产构建面。
cmake_minimum_required(VERSION 3.16)
project(astrocs_v6_real_science LANGUAGES CXX)

set(RS_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(RS_REPO "${RS_DIR}/../../../.." ABSOLUTE)

add_library(astrocs_v6_real_science_kernel STATIC
  "${RS_REPO}/lib/dynamic_psf/src/psf_information.cpp"
  "${RS_REPO}/lib/snr_estimator/cpp/src/information_weight.cpp"
  "${RS_REPO}/lib/photometric_calib/cpp/src/psfsw.cpp")
target_include_directories(astrocs_v6_real_science_kernel PUBLIC
  "${RS_REPO}/lib/dynamic_psf/include"
  "${RS_REPO}/lib/snr_estimator/include"
  "${RS_REPO}/lib/photometric_calib/include"
  "${RS_REPO}/lib/common"
  "${RS_REPO}/third_party")
set_target_properties(astrocs_v6_real_science_kernel PROPERTIES
  CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON)
if(UNIX)
  target_link_libraries(astrocs_v6_real_science_kernel PUBLIC m)
endif()

add_executable(v6_real_science_driver v6_real_science_driver.cpp)
target_link_libraries(v6_real_science_driver PRIVATE astrocs_v6_real_science_kernel)
set_target_properties(v6_real_science_driver PROPERTIES
  CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)

add_executable(v6_real_science_negative "${RS_DIR}/../oracle/v6_real_science_negative.cpp")
target_include_directories(v6_real_science_negative PRIVATE "${RS_REPO}/cli")
target_link_libraries(v6_real_science_negative PRIVATE astrocs_v6_real_science_kernel)
set_target_properties(v6_real_science_negative PROPERTIES
  CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
